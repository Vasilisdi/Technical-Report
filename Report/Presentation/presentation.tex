	\section{Introduction}
	\subsection{Predictive Maintenance and Condition Monitoring}
	\begin{frame}{Predictive Maintenance and Condition Monitoring}
		\begin{itemize}[<+-| alert@+>]
			\item Aims at tracking the health status of machines.
			\item Tracks the deterioration of temperature and vibration.
			\item Alert maintenance to take action before it is to late.		
		\end{itemize}
	\end{frame}
	
	\subsection{Problem Requirements}
	\begin{frame}{Problem Requirements}
		\begin{itemize}[<+-| alert@+>]
			\item Sensing entity for data collection.
			\item Processing unit for data analysis.
			\item Space for data storage.
			\item Cloud-based for remote access.		
		\end{itemize}
	\end{frame}

	\subsection{Objective of the Thesis}
	\begin{frame}{Objective of the Thesis}
	    \begin{itemize}[<+-| alert@+>]
			\item Development of a low-cost embedded system for predictive maintenance [see Figure \ref{fig:System}].
			\item Development of software for data acquisition and transmission from Arduino UNO R3 to Raspberry Pi 4.
			\item Development of software for FFT-based spectral analysis software on Raspberry Pi 4 for frequency-domain feature extraction.
			\item Development of a secure, authenticated API client for structured data transmission from Raspberry Pi 4 to Supabase.
			\item Implementation of a Vercel API endpoint that fetches measurement data from Supabase database.
			\item Development of a React frontend that requests and visualizes data from the Vercel API.
		\end{itemize}
	\end{frame}

	\begin{frame}{System boundaries}
		\begin{itemize}[<+-| alert@+>]
			\item The ADXL335 measurement range is $\pm 3\,g$.
			\item The Arduino UNO R3 sampling rate is 700 [Hz].
			\item Flash memory and SRAM limitations require data to be transmitted in batches.
			\item The frequency-domain analysis is limited to 350 Hz (Nyquist limit).
			\item The measurement time window determines the total number of samples collected.
		\end{itemize}
	\end{frame}

	\section{Results}
	\begin{frame}{System Verification and Validation}
		\begin{itemize}[<+-| alert@+>]
			\item End-to-end checks were performed across the entire system: data acquisition from ADXL335 sensors via Arduino UNO, serial transmission to Raspberry Pi 4, processing and posting to Supabase, and frontend visualization in Vercel/React.
			\item Stress tests confirmed reliable operation under expected data loads.
			\item Verified no data loss, correct timing, and accurate real-time plotting.
			\item System performance validated for FFT processing, transmission rates, and visualization updates.
		\end{itemize}
	\end{frame}

	\section{Conclusions}
	\begin{frame}{System Overview and Achievements}
		\begin{itemize}[<+-| alert@+>]
			\item Developed and validated an end-to-end condition monitoring system for bearing fault detection using vibration analysis.
			\item Integrated hardware (ADXL335, Arduino UNO, Raspberry Pi 4) with cloud services (Supabase, Vercel) for data acquisition, processing, and visualization.
			\item FFT-based comparative analysis of X, Y, Z axes and magnitude against healthy baselines for fault identification.
			\item Arduino collects and transmits triaxial acceleration in batches at 700 [Hz] sampling rate without loss; Raspberry Pi converts to frequency-domain and posts to Supabase.
			\item Vercel dashboard visualizes frequency spectra interactively for real-time monitoring.
		\end{itemize}
	\end{frame}
